sk.java
import javax.servlet.*;
import javax.servlet.http.*;
import java.io.*;
import java.util.* ;

public class sk   extends HttpServlet
{
    public void doGet(HttpServletRequest req,
                      HttpServletResponse resp)
        throws ServletException, java.io.IOException
    {
        PrintWriter out = resp.getWriter();
                    if (req.isUserInRole("shankar"))
            out.print("u r logged in via servlet by the user shankar");
            else if  (req.isUserInRole("mohan"))
          out.print("u r logged in via servlet by the user mohan");
    }
}
