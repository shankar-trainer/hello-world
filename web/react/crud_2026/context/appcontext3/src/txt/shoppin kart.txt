The reducer function specifies how the state changes in response to actions like 'ADD_TO_CART' or 'REMOVE_FROM_CART'.

export const CartActions = {
  ADD_TO_CART: 'ADD_TO_CART',
  REMOVE_FROM_CART: 'REMOVE_FROM_CART',
  INCREMENT_QUANTITY: 'INCREMENT_QUANTITY',
  DECREMENT_QUANTITY: 'DECREMENT_QUANTITY',
};

export const cartReducer = (state, action) => {
  switch (action.type) {
    case CartActions.ADD_TO_CART:
      // Logic to add a new item or increment quantity if it exists
      const existingItem = state.items.find(item => item.id === action.payload.id);
      if (existingItem) {
        return {
          ...state,
          items: state.items.map(item =>
            item.id === action.payload.id
              ? { ...item, quantity: item.quantity + 1 }
              : item
          ),
        };
      } else {
        return {
          ...state,
          items: [...state.items, { ...action.payload, quantity: 1 }],
        };
      }
    case CartActions.REMOVE_FROM_CART:
      // Logic to remove an item completely
      return {
        ...state,
        items: state.items.filter(item => item.id !== action.payload.id),
      };
    // Add logic for INCREMENT_QUANTITY and DECREMENT_QUANTITY
    default:
      return state;
  }
};

2. CartContext.js (Context and Provider)
This file creates the context and provides the state and dispatch function to the entire application.

javascript
import React, { createContext, useReducer, useContext } from 'react';
import { cartReducer } from './cartReducer';

const CartContext = createContext();

const initialState = {
  items: [],
};

export const CartProvider = ({ children }) => {
  const [state, dispatch] = useReducer(cartReducer, initialState);

  return (
    <CartContext.Provider value={{ state, dispatch }}>
      {children}
    </CartContext.Provider>
  );
};

export const useCart = () => {
  return useContext(CartContext);
};

4. ProductItem.js (Using the Dispatch function) 
Components can use the useCart hook to access the dispatch function and update the state. 
javascript
import React from 'react';
import { useCart, CartActions } from './CartContext';

const ProductItem = ({ product }) => {
  const { dispatch } = useCart();

  const addToCartHandler = () => {
    dispatch({ type: CartActions.ADD_TO_CART, payload: product });
  };

  return (
    <div>
      <h3>{product.name}</h3>
      <p>${product.price}</p>
      <button onClick={addToCartHandler}>Add to Cart</button>
    </div>
  );
};

export default ProductItem;

import React from 'react';
import { useCart } from './CartContext';

const Cart = () => {
  const { state } = useCart();
  
  const total = state.items.reduce((sum, item) => sum + item.price * item.quantity, 0);

  return (
    <div>
      <h2>Your Cart</h2>
      {state.items.length === 0 ? (
        <p>Cart is empty</p>
      ) : (
        <ul>
          {state.items.map(item => (
            <li key={item.id}>
              {item.name} - Quantity: {item.quantity}
            </li>
          ))}
        </ul>
      )}
      <h3>Total: ${total.toFixed(2)}</h3>
    </div>
  );
};

export default Cart;
