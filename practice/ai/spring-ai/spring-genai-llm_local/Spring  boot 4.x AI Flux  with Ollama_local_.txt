Step 4: Write the Chat Controller
Leverage Spring AI’s modern, fluent ChatClient API. Instead of injecting the model directly, Spring recommends initializing a thread-safe client using the auto-configured ChatClient.Builder.

 +2
java
package com.example.ai.controller;

import org.springframework.ai.chat.client.ChatClient;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;
import reactor.core.publisher.Flux;

@RestController
@RequestMapping("/api/v1/ai")
public class LocalAiController {

    private final ChatClient chatClient;

    // Constructor injection guarantees the client setup using Spring Boot's autowired builder
    public LocalAiController(ChatClient.Builder chatClientBuilder) {
        this.chatClient = chatClientBuilder.build();
    }

    /**
     * Standard block response endpoint
     */
    @GetMapping("/ask")
    public String askLocalModel(@RequestParam(value = "message") String message) {
        return this.chatClient.prompt()
                .user(message)
                .call()
                .content();
    }

    /**
     * Highly recommended for local models: Streamed tokens as they generate
     */
    @GetMapping(value = "/stream", produces = "text/event-stream")
    public Flux<String> streamLocalModel(@RequestParam(value = "message") String message) {
        return this.chatClient.prompt()
                .user(message)
                .stream()
                .content();
    }
}
