 application:
   name: api-gateway
 cloud:
   gateway:
     server:
       webmvc:
         routes:
           - id: example_route

               - Path=/example/**

             '[/**]':
               allowedOrigins:
                 - "https://yourfrontenddomain.com"
               allowedMethods:
                 - GET
                 - POST
                 - PUT
                 - DELETE
                 - OPTIONS
               allowedHeaders:
                 - "*"
               allowCredentials: true
               maxAge: 3600
